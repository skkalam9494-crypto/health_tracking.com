  storageBucket: "YOUR_PROJECT.appspot.com",
  messagingSenderId: "YOUR_SENDER_ID",
  appId: "YOUR_APP_ID"
};
firebase.initializeApp(firebaseConfig);
const auth = firebase.auth();
const db = firebase.firestore();

// 🔹 Auth Buttons
document.getElementById("signInBtn").onclick = ()=>{
  auth.signInWithPopup(new firebase.auth.GoogleAuthProvider()).then(res=>alert("Signed in as: "+res.user.displayName));
};
document.getElementById("signOutBtn").onclick = ()=> auth.signOut().then(()=>alert("Signed out"));

// 🔹 User Data
let steps=0, calories=0, weightHistory=[], currentUser=null;

auth.onAuthStateChanged(user=>{
  if(user){
    currentUser=user;
    const userRef=db.collection("users").doc(user.uid);
    userRef.get().then(doc=>{
      if(doc.exists){
        steps=doc.data().steps||0;
        calories=doc.data().calories||0;
        weightHistory=doc.data().weightHistory||[];
      }
      renderAll();
    });
  } else currentUser=null;
});

// 🔹 Steps
document.getElementById("addStepsBtn").onclick = ()=>{ steps+=1000; saveData("steps",steps); renderSteps(); };
document.getElementById("resetStepsBtn").onclick = ()=>{ if(confirm("Reset steps?")){ steps=0; saveData("steps",steps); renderSteps(); } };
function renderSteps(){ let pct=Math.min((steps/10000)*100,100); const p=document.getElementById("stepsProgress"); p.style.width=pct+"%"; p.innerText=Math.round(pct)+"%"; }

// 🔹 Heartbeat
document.getElementById("heartbeatBtn").onclick = ()=> document.getElementById("heartbeatValue").innerText = Math.floor(Math.random()*60 +60);

// 🔹 Calories
document.getElementById("addCaloriesBtn").onclick = ()=>{ calories+=200; saveData("calories",calories); renderCalories(); };
document.getElementById("resetCaloriesBtn").onclick = ()=>{ if(confirm("Reset calories?")){ calories=0; saveData("calories",calories); renderCalories(); } };
function renderCalories(){
  document.getElementById("caloriesValue").innerText = calories;
  let ctx = document.getElementById("caloriesChart").getContext("2d");
  ctx.clearRect(0,0,150,150);
  ctx.beginPath(); ctx.arc(75,75,60,0,2*Math.PI); ctx.strokeStyle="#ddd"; ctx.lineWidth=15; ctx.stroke();
  let pct=Math.min(calories/2000,1);
  ctx.beginPath(); ctx.arc(75,75,60,-0.5*Math.PI,(pct*2*Math.PI)-0.5*Math.PI); ctx.strokeStyle="#ff4500"; ctx.lineWidth=15; ctx.stroke();
}

// 🔹 Weight
document.getElementById("addWeightBtn").onclick = ()=>{
  let val=document.getElementById("weightInput").value;
  if(val){ weightHistory.push(val); saveData("weightHistory",weightHistory); renderWeight(); document.getElementById("weightInput").value=""; }
};
document.getElementById("resetWeightBtn").onclick = ()=>{ if(confirm("Reset weight?")){ weightHistory=[]; saveData("weightHistory",weightHistory); renderWeight(); } };
function renderWeight(){ const list=document.getElementById("weightHistory"); list.innerHTML=""; weightHistory.forEach(w=>{ let li=document.createElement("li"); li.innerText=w+" kg"; list.appendChild(li); }); }

// 🔹 Save Data to Firebase
function saveData(field,value){ if(currentUser) db.collection("users").doc(currentUser.uid).set({[field]:value},{merge:true}); }
